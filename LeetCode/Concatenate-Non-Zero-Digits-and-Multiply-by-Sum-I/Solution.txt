class Solution {
public:
    long long sumAndMultiply(int n) {
    string s = to_string(n);
    string str = "";
    long long sum = 0;

    for (char c : s) {
        if (c != '0') {
            str += c;
            sum += (c - '0');
        }
    }

    long long a = 0;
    if (!str.empty()) {
        a = stoll(str);
    }

    return a * sum;
       
    }
};
