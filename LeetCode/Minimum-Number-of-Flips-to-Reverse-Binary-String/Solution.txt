class Solution {
public:
    int minimumFlips(int n) {
        string value = "";
        if(n == 0 ) return 0;
        else{
            while(n>0){
                value = char('0' + (n%2))+value;
                n /= 2;
            }
        }
        string check = value;
        reverse(check.begin(),check.end());
        int cnt = 0;
        for(int i = 0 ;i<value.size();i++){
            if(value[i] != check[i]){
                cnt++;
            }
        }return cnt;
    }
};
