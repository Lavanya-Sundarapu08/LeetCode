class Solution {
public:
    int countMajoritySubarrays(vector<int>& nums, int target) {
        int n = nums.size();
        int cnt= 0;
        for(int i =0;i<n;i++){
            int tcnt = 0;
            int sum =0;
        for(int j =i;j<n;++j){
            sum++;
            if(nums[j] == target ) tcnt++;
            if(tcnt > (sum/2)){
                cnt++;
            }
        }
        }
        return cnt;
    }
};
